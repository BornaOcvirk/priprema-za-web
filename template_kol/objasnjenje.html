2. Detaljan pregled svih izmjena po datotekama

📁 template/server/todos.js (Poslužiteljska baza podataka)

Što je promijenjeno:

  • Uvezen je randomUUID iz ugrađenog Node.js modula crypto.
  • Svakom zapisu u nizu todos generiran je jedinstveni UUID atribut: id: randomUUID().
  • Niz je eksportiran: export { todos }; export default todos;.

Zašto:

U originalnom predlošku zapisi nisu imali identifikatore niti je modul bio eksportiran. Za ažuriranje i brisanje točno određenog zapisa, poslužitelj i klijent moraju komunicirati putem jedinstvenog ključa (UUID). Ovo donosi 2 boda za UUID i omogućuje uvoz u poslužiteljski modul.

import { randomUUID } from "crypto";

const todos = [
  { id: randomUUID(), title: "Buy groceries", timestamp: 1764845228329, done: true },
  { id: randomUUID(), title: "Email client",  timestamp: 1764845228329, done: true },
  ...
];

export { todos };
export default todos;
📁 template/server/index.js (Glavni modul poslužitelja)

Što je promijenjeno:

  • Uvezen niz todos iz ./todos.js.
  • Dodan middleware app.use(express.json()); za prihvaćanje JSON tijela zahtjeva.
  • Definirana ruta za dohvaćanje: GET /api/todos.
  • Definirana ruta za uređivanje: PUT /api/todos/:id (traži element po UUID-u i ažurira done).
  • Definirana ruta za brisanje: DELETE /api/todos/:id (uklanja zapis pomoću splice).

Zašto:

Zadatak 1 zahtijeva REST API rute za dohvat, uređivanje i brisanje. express.json() je ključan jer klijentska toggle funkcija šalje JSON tijelo ({ done: true/false }) koje Express bez ovog parsera ne može pročitati.

import express from 'express';
import todos from './todos.js';

const app = express();
app.use(express.json());
app.use(express.urlencoded({ extended: false }));

// Dohvaćanje zapisa
app.get("/api/todos", (req, res) => res.json(todos));

// Uređivanje postojećeg zapisa (toggle statusa)
app.put("/api/todos/:id", (req, res) => {
    const todo = todos.find(t => t.id === req.params.id);
    if (!todo) return res.status(404).json({ error: "Not found" });
    if (req.body.done !== undefined) todo.done = req.body.done;
    res.json(todo);
});

// Brisanje postojećeg zapisa
app.delete("/api/todos/:id", (req, res) => {
    const index = todos.findIndex(t => t.id === req.params.id);
    if (index === -1) return res.status(404).json({ error: "Not found" });
    const [removed] = todos.splice(index, 1);
    res.json(removed);
});

📁 template/src/js/index.js (Klijentska logika, animacije i API pozivi)

Što je implementirano:

  • format_date(timestamp): pretvara Unix milisekunde u format M/D/YYYY (npr. 12/4/2025).
  • get_todos(callback): asinkroni dohvat podataka preko jQueryja ($.get('/api/todos')) koji vraća Promise.
  • toggle(id): šalje PUT /api/todos/${id} sa suprotnim done statusom i mijenja ikonu (check-circle-fill ↔ x-circle-fill).
  • remove(id): šalje DELETE /api/todos/${id} i uklanja karticu uz .fadeOut(300) animaciju.
  • render_todos(todos): stvara DOM elemente kartica, povezuje event listenere i pokreće .fadeIn(500) pri učitavanju.
  • Globalno izlaganje: funkcije su postavljene na window objekt radi automatskog testiranja.
import $ from "jquery";

function format_date(timestamp) {
    const d = new Date(timestamp);
    return `${d.getMonth() + 1}/${d.getDate()}/${d.getFullYear()}`;
}

function get_todos(callback) {
    const request = $.get("/api/todos");
    if (typeof callback === "function") request.then(callback);
    return request;
}

function toggle(id) {
    const $col = $(`[data-id="${id}"]`);
    const $statusIcon = $col.find(".status-icon");
    const isDone = $statusIcon.hasClass("bi-check-circle-fill");
    const newStatus = !isDone;

    return $.ajax({
        url: `/api/todos/${id}`,
        method: "PUT",
        contentType: "application/json",
        data: JSON.stringify({ done: newStatus }),
        success: (updated) => {
            if (updated.done) {
                $statusIcon.removeClass("bi-x-circle-fill text-danger").addClass("bi-check-circle-fill text-success");
            } else {
                $statusIcon.removeClass("bi-check-circle-fill text-success").addClass("bi-x-circle-fill text-danger");
            }
        }
    });
}

function remove(id) {
    const $col = $(`[data-id="${id}"]`);
    return $.ajax({
        url: `/api/todos/${id}`,
        method: "DELETE",
        success: () => {
            $col.fadeOut(300, function () { $(this).remove(); });
        }
    });
}

function render_todos(todos) {
    const $container = $("#todos-container");
    $container.empty();

    todos.forEach((todo) => {
        const formattedDate = format_date(todo.timestamp);
        const iconClass = todo.done ? "bi-check-circle-fill text-success" : "bi-x-circle-fill text-danger";

        const $col = $(`
            <div class="col" data-id="${todo.id}">
                <div class="card rounded-4 border shadow-sm p-3 h-100">
                    <div class="card-body d-flex flex-column justify-content-between p-0">
                        <div>
                            <h5 class="card-title fw-bold mb-2">${todo.title}</h5>
                            <p class="card-text mb-1"><span class="fw-bold">When:</span> ${formattedDate}</p>
                            <p class="card-text mb-3"><span class="fw-bold">Status:</span> <i class="bi ${iconClass} status-icon"></i></p>
                        </div>
                        <div class="d-flex gap-2 mt-auto">
                            <button class="btn btn-outline-secondary rounded-pill flex-fill btn-toggle">Toggle</button>
                            <button class="btn btn-outline-secondary rounded-pill flex-fill btn-remove">Remove</button>
                        </div>
                    </div>
                </div>
            </div>
        `);

        $col.find(".btn-toggle").on("click", () => toggle(todo.id));
        $col.find(".btn-remove").on("click", () => remove(todo.id));

        $col.hide();
        $container.append($col);
        $col.fadeIn(500);
    });
}

window.format_date = format_date;
window.get_todos = get_todos;
window.toggle = toggle;
window.remove = remove;
window.render_todos = render_todos;

$(() => {
    get_todos().then((todos) => render_todos(todos));
});
