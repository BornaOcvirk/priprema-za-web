2. Detaljna objašnjenja po datotekama

📁 server/books.js (Generiranje UUID-a i izvoz)

Rješenje i ključne točke:

  • Koristi se ugrađeni Node.js modul crypto: import { randomUUID } from "crypto";
  • Svakom objektu dodaje se id: randomUUID() kako bi klijent i poslužitelj mogli adresirati specifičan resurs u PUT i DELETE zahtjevima.
  • Modul se eksportira s export { books }; export default books;.
import { randomUUID } from "crypto";

const books = [
  { id: randomUUID(), title: "Clean Code", timestamp: 1765018057945, read: true },
  { id: randomUUID(), title: "The Pragmatic Programmer", timestamp: 1764845228329, read: true },
  { id: randomUUID(), title: "Eloquent JavaScript", timestamp: 1764931675258, read: false },
  ...
];

export { books };
export default books;
📁 server/index.js (REST API rute)

Rješenje i ključne točke:

  • app.use(express.json()); je obavezan middleware kako bi Express mogao pročitati JSON tijelo koje šalje klijentski PUT poziv ({ read: true/false }).
  • U ruti za ažuriranje (PUT /api/books/:id) provjeravamo req.body.read !== undefined i spremamo novu vrijednost.
  • U ruti za brisanje (DELETE /api/books/:id) pronalazimo indeks elementa i uklanjamo ga pomoću books.splice(index, 1).
import express from 'express'
import books from './books.js'

const app = express()
app.use(express.json());
app.use(express.urlencoded({ extended: false }));

// GET
app.get("/api/books", (req, res) => res.json(books));

// PUT
app.put("/api/books/:id", (req, res) => {
    const item = books.find((e) => e.id === req.params.id);
    if (!item) return res.status(404).json({ error: "Book not found" });
    if (req.body.read !== undefined) item.read = req.body.read;
    return res.json(item);
});

// DELETE
app.delete("/api/books/:id", (req, res) => {
    const index = books.findIndex((e) => e.id === req.params.id);
    if (index === -1) return res.status(404).json({ error: "Book not found" });
    const [removed] = books.splice(index, 1);
    return res.json(removed);
});

app.listen(3000, () => console.log('Server is running on http://localhost:3000'));

📁 src/js/index.js (Dinamičko renderiranje i animacije)

Svaka kartica dobiva data-id="${item.id}" atribut kako bi funkcije toggle i remove mogle točno pronaći odgovarajući stupac u DOM-u putem $(`[data-id="${id}"]`).

import $ from "jquery";

function format_date(timestamp) {
    const d = new Date(timestamp);
    return `${d.getMonth() + 1}/${d.getDate()}/${d.getFullYear()}`;
}

function get_books(callback) {
    const request = $.get("/api/books");
    if (typeof callback === "function") request.then(callback);
    return request;
}

function toggle(id) {
    const $col = $(`[data-id="${id}"]`);
    const $statusIcon = $col.find(".status-icon");
    const isRead = $statusIcon.hasClass("bi-bookmark-check-fill");
    const newStatus = !isRead;

    return $.ajax({
        url: `/api/books/${id}`,
        method: "PUT",
        contentType: "application/json",
        data: JSON.stringify({ read: newStatus }),
        success: (updated) => {
            if (updated.read) {
                $statusIcon
                    .removeClass("bi-bookmark-x-fill text-danger")
                    .addClass("bi-bookmark-check-fill text-success");
            } else {
                $statusIcon
                    .removeClass("bi-bookmark-check-fill text-success")
                    .addClass("bi-bookmark-x-fill text-danger");
            }
        }
    });
}

function remove(id) {
    const $col = $(`[data-id="${id}"]`);
    return $.ajax({
        url: `/api/books/${id}`,
        method: "DELETE",
        success: () => {
            $col.fadeOut(300, function () { $(this).remove(); });
        }
    });
}

function render_books(books) {
    const $container = $("#books-container");
    $container.empty();

    books.forEach((item) => {
        const formattedDate = format_date(item.timestamp);
        const iconClass = item.read
            ? "bi-bookmark-check-fill text-success"
            : "bi-bookmark-x-fill text-danger";

        const $col = $(`
            <div class="col" data-id="${item.id}">
                <div class="card rounded-4 border shadow-sm p-3 h-100">
                    <div class="card-body d-flex flex-column justify-content-between p-0">
                        <div>
                            <h5 class="card-title fw-bold mb-2">${item.title}</h5>
                            <p class="card-text mb-1"><span class="fw-bold">Added:</span> ${formattedDate}</p>
                            <p class="card-text mb-3"><span class="fw-bold">Status:</span> <i class="bi ${iconClass} status-icon"></i></p>
                        </div>
                        <div class="d-flex gap-2 mt-auto">
                            <button class="btn btn-outline-secondary rounded-pill flex-fill btn-toggle">Toggle</button>
                            <button class="btn btn-outline-secondary rounded-pill flex-fill btn-remove">Remove</button>
                        </div>
                    </div>
                </div>
            </div>
        `);

        $col.find(".btn-toggle").on("click", () => toggle(item.id));
        $col.find(".btn-remove").on("click", () => remove(item.id));

        $col.hide();
        $container.append($col);
        $col.fadeIn(500);
    });
}

window.format_date = format_date;
window.get_books = get_books;
window.toggle = toggle;
window.remove = remove;
window.render_books = render_books;

export { format_date, get_books, toggle, remove, render_books };

$(() => {
    get_books().then((books) => render_books(books));
});
