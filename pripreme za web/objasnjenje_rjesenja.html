2. Detaljna objašnjenja po datotekama

📁 server/events.js (Generiranje UUID-a i izvoz)

Rješenje i ključne točke:

  • Koristi se ugrađeni Node.js modul crypto: import { randomUUID } from "crypto";
  • Svakom objektu dodaje se id: randomUUID() kako bi klijent i poslužitelj mogli adresirati specifičan resurs u PUT i DELETE zahtjevima.
  • Modul se eksportira s export { events }; export default events;.
import { randomUUID } from "crypto";

const events = [
  { id: randomUUID(), title: "Web Dev Workshop", timestamp: 1765018057945, confirmed: true },
  { id: randomUUID(), title: "Design Sprint", timestamp: 1764845228329, confirmed: true },
  { id: randomUUID(), title: "Cybersecurity Meetup", timestamp: 1764931675258, confirmed: false },
  ...
];

export { events };
export default events;
📁 server/index.js (REST API rute)

Rješenje i ključne točke:

  • app.use(express.json()); je obavezan middleware kako bi Express mogao pročitati JSON tijelo koje šalje klijentski PUT poziv ({ confirmed: true/false }).
  • U ruti za ažuriranje (PUT /api/events/:id) provjeravamo req.body.confirmed !== undefined i spremamo novu vrijednost.
  • U ruti za brisanje (DELETE /api/events/:id) pronalazimo indeks elementa i uklanjamo ga pomoću events.splice(index, 1).
import express from 'express'
import events from './events.js'

const app = express()
app.use(express.json());
app.use(express.urlencoded({ extended: false }));

// GET
app.get("/api/events", (req, res) => res.json(events));

// PUT
app.put("/api/events/:id", (req, res) => {
    const item = events.find((e) => e.id === req.params.id);
    if (!item) return res.status(404).json({ error: "Event not found" });
    if (req.body.confirmed !== undefined) item.confirmed = req.body.confirmed;
    return res.json(item);
});

// DELETE
app.delete("/api/events/:id", (req, res) => {
    const index = events.findIndex((e) => e.id === req.params.id);
    if (index === -1) return res.status(404).json({ error: "Event not found" });
    const [removed] = events.splice(index, 1);
    return res.json(removed);
});

app.listen(3000, () => console.log('Server is running on http://localhost:3000'));

📁 src/js/index.js (Dinamičko renderiranje i animacije)

Svaka kartica dobiva data-id="${item.id}" atribut kako bi funkcije toggle i remove mogle točno pronaći odgovarajući stupac u DOM-u putem $(`[data-id="${id}"]`).

import $ from "jquery";

function format_date(timestamp) {
    const d = new Date(timestamp);
    return `${d.getMonth() + 1}/${d.getDate()}/${d.getFullYear()}`;
}

function get_events(callback) {
    const request = $.get("/api/events");
    if (typeof callback === "function") request.then(callback);
    return request;
}

function toggle(id) {
    const $col = $(`[data-id="${id}"]`);
    const $statusIcon = $col.find(".status-icon");
    const isConfirmed = $statusIcon.hasClass("bi-calendar-check-fill");
    const newStatus = !isConfirmed;

    return $.ajax({
        url: `/api/events/${id}`,
        method: "PUT",
        contentType: "application/json",
        data: JSON.stringify({ confirmed: newStatus }),
        success: (updated) => {
            if (updated.confirmed) {
                $statusIcon
                    .removeClass("bi-calendar-x-fill text-danger")
                    .addClass("bi-calendar-check-fill text-success");
            } else {
                $statusIcon
                    .removeClass("bi-calendar-check-fill text-success")
                    .addClass("bi-calendar-x-fill text-danger");
            }
        }
    });
}

function remove(id) {
    const $col = $(`[data-id="${id}"]`);
    return $.ajax({
        url: `/api/events/${id}`,
        method: "DELETE",
        success: () => {
            $col.fadeOut(300, function () { $(this).remove(); });
        }
    });
}

function render_events(events) {
    const $container = $("#events-container");
    $container.empty();

    events.forEach((item) => {
        const formattedDate = format_date(item.timestamp);
        const iconClass = item.confirmed
            ? "bi-calendar-check-fill text-success"
            : "bi-calendar-x-fill text-danger";

        const $col = $(`
            <div class="col" data-id="${item.id}">
                <div class="card rounded-4 border shadow-sm p-3 h-100">
                    <div class="card-body d-flex flex-column justify-content-between p-0">
                        <div>
                            <h5 class="card-title fw-bold mb-2">${item.title}</h5>
                            <p class="card-text mb-1"><span class="fw-bold">When:</span> ${formattedDate}</p>
                            <p class="card-text mb-3"><span class="fw-bold">Status:</span> <i class="bi ${iconClass} status-icon"></i></p>
                        </div>
                        <div class="d-flex gap-2 mt-auto">
                            <button class="btn btn-outline-secondary rounded-pill flex-fill btn-toggle">Toggle</button>
                            <button class="btn btn-outline-secondary rounded-pill flex-fill btn-remove">Remove</button>
                        </div>
                    </div>
                </div>
            </div>
        `);

        $col.find(".btn-toggle").on("click", () => toggle(item.id));
        $col.find(".btn-remove").on("click", () => remove(item.id));

        $col.hide();
        $container.append($col);
        $col.fadeIn(500);
    });
}

window.format_date = format_date;
window.get_events = get_events;
window.toggle = toggle;
window.remove = remove;
window.render_events = render_events;

export { format_date, get_events, toggle, remove, render_events };

$(() => {
    get_events().then((events) => render_events(events));
});
